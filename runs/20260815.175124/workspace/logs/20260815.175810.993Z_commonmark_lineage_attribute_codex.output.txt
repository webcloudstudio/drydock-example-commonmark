<requirement name="pipe-through-program" stories="cli-executable">
def pipe_through_prog(prog, text):
    p1 = Popen(prog.split(), stdout=PIPE, stdin=PIPE, stderr=PIPE)
    [result, err] = p1.communicate(input=text.encode('utf-8'))
    return [p1.returncode, result, err]
</requirement>
<requirement name="use-cmark-library" stories="html-renderer">
def use_library(lib, text):
    textbytes = text.encode('utf-8')
    textlen = len(textbytes)
    return [0, lib(textbytes, textlen, 0), '']
</requirement>
