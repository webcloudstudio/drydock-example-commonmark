"""List item tests: the seven declared checks from FEATURE-List-Items.md, then block-tree and exact-output
tests for markers, content columns, empty items, start numbers, laziness and interruption.

The declared checks are transcribed first, with their assertions unchanged. The rest act on the
block tree returned by `parse_lines` and on the exact stdout bytes, which is the state handed back.
"""

import subprocess
from pathlib import Path

import pytest

from mdparser.blocks import DEFAULT_REGISTRY, LIST_ITEM, PARAGRAPH, Block, parse_lines


def run_commonmark(project_root: Path, source: str) -> subprocess.CompletedProcess[str]:
    return subprocess.run(
        [str(project_root / "commonmark")],
        input=source,
        capture_output=True,
        text=True,
        cwd=project_root,
        check=False,
    )


def kinds(block: Block) -> list[str]:
    """Names of the child block types of `block`, in order."""
    return [child.kind.name for child in block.children]


def paragraph_texts(block: Block) -> list[str]:
    """Raw content of every paragraph under `block`, in document order."""
    texts: list[str] = []
    for child in block.children:
        if child.kind is PARAGRAPH:
            texts.append(child.content)
        texts.extend(paragraph_texts(child))
    return texts


def only_item(lines: list[str]) -> Block:
    """The single list item at the top of the document built from `lines`."""
    root = parse_lines(lines)
    assert kinds(root) == ["list_item"]
    return root.children[0]


# --- Declared acceptance checks (transcribed from FEATURE-List-Items.md) ---


def test_items_bullet_items(project_root: Path) -> None:
    """Each bullet marker line starts one li element, and consecutive items of one marker form one ul."""
    items = ["first", "second", "third"]
    source = "".join("- " + text + "\n" for text in items)
    result = subprocess.run([str(project_root / "commonmark")], input=source, capture_output=True, text=True)
    print(result.stdout)
    assert result.returncode == 0
    assert result.stdout.count("<li>") == len(items)
    assert result.stdout.count("<ul>") == 1
    for text in items:
        assert text in result.stdout


def test_items_ordered_start_number(project_root: Path) -> None:
    """An ordered list takes its start number from the first item, and ten digits is not a marker."""
    number = 123456789
    good = subprocess.run([str(project_root / "commonmark")], input=str(number) + ". ok\n", capture_output=True, text=True)
    bad = subprocess.run([str(project_root / "commonmark")], input=str(number) + "0. not ok\n", capture_output=True, text=True)
    print(good.stdout)
    print(bad.stdout)
    assert good.returncode == 0
    assert bad.returncode == 0
    assert ('start="' + str(number) + '"') in good.stdout
    assert bad.stdout.count("<ol") == 0


def test_items_continuation_paragraph(project_root: Path) -> None:
    """A continuation paragraph indented to the content column joins the item, giving a loose item with two paragraphs."""
    result = subprocess.run([str(project_root / "commonmark")], input="- one\n\n  two\n", capture_output=True, text=True)
    print(result.stdout)
    assert result.returncode == 0
    assert result.stdout.count("<li>") == 1
    assert result.stdout.count("<p>") == 2


def test_items_insufficient_indent_ends_item(project_root: Path) -> None:
    """A paragraph indented less than the content column is outside the item."""
    result = subprocess.run([str(project_root / "commonmark")], input="- one\n\n two\n", capture_output=True, text=True)
    print(result.stdout)
    assert result.returncode == 0
    assert result.stdout.count("<li>") == 1
    assert result.stdout.count("<p>") == 1


def test_items_empty_items(project_root: Path) -> None:
    """Empty items are allowed in the middle of a list, but an empty item cannot interrupt a paragraph."""
    middle = subprocess.run([str(project_root / "commonmark")], input="- foo\n-\n- bar\n", capture_output=True, text=True)
    interrupt = subprocess.run([str(project_root / "commonmark")], input="foo\n*\n", capture_output=True, text=True)
    print(middle.stdout)
    print(interrupt.stdout)
    assert middle.returncode == 0
    assert interrupt.returncode == 0
    assert middle.stdout.count("<li>") == 2 + 1
    assert interrupt.stdout.count("<li>") == 0


def test_items_nested_and_code(project_root: Path) -> None:
    """A sublist indented to the content column nests, and a code block six spaces in is code inside the item."""
    nested = subprocess.run([str(project_root / "commonmark")], input="- foo\n  - bar\n", capture_output=True, text=True)
    code = subprocess.run([str(project_root / "commonmark")], input="- foo\n\n      bar\n", capture_output=True, text=True)
    print(nested.stdout)
    print(code.stdout)
    assert nested.returncode == 0
    assert code.returncode == 0
    assert nested.stdout.count("<ul>") == 2
    assert code.stdout.count("<pre>") == 1


def test_items_four_spaces_is_code(project_root: Path) -> None:
    """A marker indented four spaces is code, not a list item."""
    result = subprocess.run([str(project_root / "commonmark")], input="    1.  A paragraph\n", capture_output=True, text=True)
    print(result.stdout)
    assert result.returncode == 0
    assert result.stdout.count("<ol") == 0
    assert result.stdout.count("<pre>") == 1

@pytest.mark.parametrize("bullet", ["-", "+", "*"])
def test_each_bullet_character_starts_an_item(bullet: str) -> None:
    item = only_item([f"{bullet} foo"])
    assert item.item is not None
    assert item.item.ordered is False
    assert item.item.delimiter == bullet


@pytest.mark.parametrize("delimiter", [".", ")"])
def test_ordered_marker_records_its_number_and_delimiter(delimiter: str) -> None:
    item = only_item([f"3{delimiter} foo"])
    assert item.item is not None
    assert item.item.ordered is True
    assert item.item.start == 3
    assert item.item.delimiter == delimiter


def test_nine_digit_number_is_a_marker() -> None:
