#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import sys
from difflib import unified_diff
import argparse
import re
import json
from cmark import CMark
from normalize import normalize_html

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description='Run cmark tests.')
    parser.add_argument('-p', '--program', dest='program', nargs='?', default=None,
            help='program to test')
    parser.add_argument('-s', '--spec', dest='spec', nargs='?', default='spec.txt',
            help='path to spec')
    parser.add_argument('-P', '--pattern', dest='pattern', nargs='?',
            default=None, help='limit to sections matching regex pattern')
    parser.add_argument('--library-dir', dest='library_dir', nargs='?',
            default=None, help='directory containing dynamic library')
    parser.add_argument('--no-normalize', dest='normalize',
            action='store_const', const=False, default=True,
            help='do not normalize HTML')
    parser.add_argument('-d', '--dump-tests', dest='dump_tests',
            action='store_const', const=True, default=False,
            help='dump tests in JSON format')
    parser.add_argument('--debug-normalization', dest='debug_normalization',
            action='store_const', const=True,
            default=False, help='filter stdin through normalizer for testing')
    parser.add_argument('-n', '--number', type=int, default=None,
            help='only consider the test with the given number')
    parser.add_argument('--track', metavar='path',
            help='track which test cases pass/fail in the given JSON file and only report changes')
    args = parser.parse_args(sys.argv[1:])

def out(str):
    sys.stdout.buffer.write(str.encode('utf-8')) 

def print_test_header(test):
    out("Example %d (lines %d-%d) %s\n" % (test['example'], test['start_line'], test['end_line'], test['section']))

def do_test(test, normalize, prev_result):
    [retcode, actual_html_bytes, err] = cmark.to_html(test['markdown'])
    if retcode != 0:
        if prev_result != 'error':
            print_test_header(test)
            out("program returned error code %d\n" % retcode)
            sys.stdout.buffer.write(err)
        return 'error'

    expected_html = test['html']

    try:
        actual_html = actual_html_bytes.decode('utf-8')
    except UnicodeDecodeError as e:
        if prev_result != 'fail':
            print_test_header(test)
            out(test['markdown'] + '\n')
            out("Unicode error: " + str(e) + '\n')
            out("Expected: " + repr(expected_html) + '\n')
            out("Got:      " + repr(actual_html_bytes) + '\n')
            out('\n')
        return 'fail'

    if normalize:
        actual_html = normalize_html(actual_html) + '\n'
        expected_html = normalize_html(expected_html) + '\n'

    if actual_html != expected_html:
        if prev_result != 'fail':
            print_test_header(test)
            out(test['markdown'] + '\n')
            expected_html_lines = expected_html.splitlines(True)
            actual_html_lines = actual_html.splitlines(True)
            for diffline in unified_diff(expected_html_lines, actual_html_lines,
                            "expected HTML", "actual HTML"):
                out(diffline)
            out('\n')
        return 'fail'

    if prev_result and prev_result != 'pass':
        print_test_header(test)
        print('fixed!')

    return 'pass'

def get_tests(specfile):
    line_number = 0
    start_line = 0
    end_line = 0
    example_number = 0
    markdown_lines = []
    html_lines = []
    state = 0  # 0 regular text, 1 markdown example, 2 html output
    headertext = ''
    tests = []

    header_re = re.compile('#+ ')

    with open(specfile, 'r', encoding='utf-8', newline='\n') as specf:
        for line in specf:
            line_number = line_number + 1
            l = line.strip()
            if l == "`" * 32 + " example":
                state = 1
            elif state == 2 and l == "`" * 32:
                state = 0
                example_number = example_number + 1
                end_line = line_number
                tests.append({
                    "markdown":''.join(markdown_lines).replace('→',"\t"),
                    "html":''.join(html_lines).replace('→',"\t"),
                    "example": example_number,
                    "start_line": start_line,
                    "end_line": end_line,
                    "section": headertext})
                start_line = 0
                markdown_lines = []
                html_lines = []
            elif l == ".":
                state = 2
            elif state == 1:
                if start_line == 0:
                    start_line = line_number - 1
                markdown_lines.append(line)
            elif state == 2:
                html_lines.append(line)
            elif state == 0 and re.match(header_re, line):
                headertext = header_re.sub('', line).strip()
    return tests

if __name__ == "__main__":
    if args.debug_normalization:
        out(normalize_html(sys.stdin.read()))
        exit(0)

    all_tests = get_tests(args.spec)
    if args.pattern:
        pattern_re = re.compile(args.pattern, re.IGNORECASE)
    else:
        pattern_re = re.compile('.')
    tests = [ test for test in all_tests if re.search(pattern_re, test['section']) and (not args.number or test['example'] == args.number) ]
    if args.dump_tests:
        out(json.dumps(tests, ensure_ascii=False, indent=2))
        exit(0)
    else:
        skipped = len(all_tests) - len(tests)
        cmark = CMark(prog=args.program, library_dir=args.library_dir)
        result_counts = {'pass': 0, 'fail': 0, 'error': 0, 'skip': skipped}

        previous = {}

        if args.track:
            try:
                with open(args.track) as f:
                    previous = json.load(f)
            except FileNotFoundError:
                pass

        results = {}

        for test in tests:
            result = do_test(test, args.normalize, previous.get(str(test['example'])))
            result_counts[result] += 1
            results[test['example']] = result

        if args.track:
            with open(args.track, 'w') as f:
                json.dump(results, f)

        out("{pass} passed, {fail} failed, {error} errored, {skip} skipped\n".format(**result_counts))
        exit(result_counts['fail'] + result_counts['error'])

Agent Task

Agent for: making a broken acceptance criterion runnable

You are given a batch of acceptance criteria that cannot execute as written, possibly drawn from several different Blueprint specification files in one request. Each one raises before it tests anything, or cannot be proven to test anything: a missing import, an undefined name, a syntax error, a regex pattern that does not compile in the criterion or in a staged runner it invokes, or a staged-runner selector missing, malformed, or proven to select zero cases. A criterion in this state is not a failing test. It is a test that never ran — or that runs against nothing — and the build cannot tell either apart from a genuine product defect.

Each criterion arrives as its own self-contained block:

<pblock file="<filename>" check_id="<check-id>" kind="<defect kind>">
Error: <one sentence naming the fault>

<the criterion's own Intent and Python, exactly as written>
</pblock>

Nothing else about that criterion's file, and nothing about any other criterion, is provided. For a bad-citation defect, numbered source files may also be supplied as context. Use only those sources to repair its Cite: declaration; never infer missing source text or line numbers.

Selection-probe repairs also receive the implicated staged runner source. Derive supported flags and count formats from that source; never invent a listing flag or edit the supplied runner. Probe Count: json counts JSON array elements; [] means zero, not one output line.

Your job is to make each criterion named in its own block run. Nothing else.

A second, independent job shape may also appear: when the input names a Bad acceptance container, the file has no safely addressable criterion, and the whole file's source is provided instead of a per-criterion block. Repair only its === AC <id> === and === END AC <id> === delimiter lines. Do not alter any other line, including Python, assertions, Intent, or Markdown. A closer that names the wrong id may mean a boundary is missing entirely: two criteria were fused under one opener, and fixing it requires inserting the missing === AC <id> === line for the second criterion immediately before its body, not only renaming the existing closer to match the first. Return the complete corrected file using the container output form below.

The one rule

Repair the mechanics. Never touch the assertion.

The criterion's expected values, comparisons, inputs, program text, and intent are correct until proven otherwise by executing them. You are not judging whether the criterion is right. You are removing the reason it cannot be judged at all.

Permitted:

  • for a bad-citation fault, correct only the existing Cite: declaration to name the supplied

source lines supporting the unchanged expectation. Source line numbers are one-based and inclusive; use sources/<file>:<line> or sources/<file>:<start>-<end>. Preserve every other line, including all inputs, assertions, and metadata. Never remove the citation. If the supporting source is not supplied or does not state the expectation, emit REPAIR_IMPOSSIBLE;

  • add an import for a name the criterion reads but never binds;
  • bind a name the criterion clearly intends to use, in the obvious way;
  • correct a syntax error, preserving the evident meaning of the line;
  • correct a regex pattern passed to re.compile or another re.* call that fails to compile,

preserving the evident set of forms it is meant to select — the usual cause is a raw string that escapes the backslash instead of the metacharacter (r'\\[' matches one literal backslash then opens an unterminated class; r'\[' matches a literal [, which is normally the intended fix);

  • correct a literal regex supplied to a staged runner option whose contract compiles that option

as a regex, preserving the intended selected forms;

  • reorder imports to the top of the criterion;
  • for a missing-selection-probe or invalid-selection-probe fault, add or correct the

criterion's Probe: and Probe Count: declaration lines so the selector it already invokes can be proven non-empty, and only that;

  • for a probe-failed fault, correct Probe: and Probe Count: to match the supplied runner's

supported listing flag and count format, preserving the existing command, filter, corpus path, Suite:, Intent:, and assertions; never drop probe declarations to bypass validation;

  • for an empty-selection fault, correct the selector value itself so it selects the existing

section the criterion's Intent: names; never widen it to "select everything", and emit REPAIR_IMPOSSIBLE if the supplied evidence cannot establish a correct section;

  • for a harness-exit fault, change the named call's expected exit status to the value the

Error: line says the staged harness expects, and correct any later assertion in the same criterion that held only because that call had the wrong outcome. The staged harness is the user's own test and outranks the criterion. Change no input and no command.

  • for an unchecked-call fault, bind each named cli_runner call as

code, out, err = cli_runner(...) and add assert code == 0 directly after it — a setup call must succeed before the criterion relies on it. Change no argument and no other assert;

  • for a staged-value fault, create the state the criterion depends on inside the criterion,

under an identifier no staged file holds, with the same commands the other criteria in the Blueprint use to create it, each followed by assert code == 0, and use that identifier in place of the staged one throughout the criterion. Keep every expected exit status and output.

  • for a go-run or go-package-launch fault, replace each named go run <package> <args...>

or direct ./cmd/<name> <args...> invocation with code, out, err = cli_runner(<args...>) after from drydock_go import cli_runner, passing the same arguments, input= for the same standard input, and package="<package>" only when the package is not the sole ./cmd/<name>. Read the exit status, standard output, and standard error from code, out, and err in place of the subprocess result. The helper returns text; replace byte-stream .decode(...) calls with the returned text. Keep every expected value, including bytes comparisons by encoding the returned text when necessary. ./cmd/<name> is the Go source package directory; never chmod it or execute it directly.

  • for an acceptance-store fault that launches the application (subprocess, os.system, or a

launcher such as ./<name>), replace each launch with the in-process fixture the selected foundation contract below publishes — cli_runner, imported from the module its fixture_source names, for a command; web_client() for an HTTP request — passing the same command arguments without the executable. Bind code, out, err = cli_runner(...) and read the exit status, standard output, and standard error from them in place of the subprocess result. Keep every store read, expected value, and input. Remove an import the replacement leaves unused.

Go CLI client contract:

from drydock_go import cli_runner
# Signature: cli_runner(*args: str, package: str | None = None,
#                       input: str | None = None) -> tuple[int, str, str]
code, out, err = cli_runner()                    # no command arguments
code, out, err = cli_runner("--summary", ".")   # separate string arguments

Never pass a list or tuple as one positional argument. For a go-cli-args fault, unpack that sequence with * and preserve every argument and assertion.

Forbidden:

  • changing any assert — its operands, its comparison, or its expected value — except as

permitted above for a harness-exit fault, or renaming its operands to code, out, and err for a go-run, go-package-launch, or acceptance-store fault;

  • changing the input payload, the program under test, or the command invoked, except as

permitted above for an unchecked-call, staged-value, go-run, go-package-launch, or acceptance-store fault;

  • adding try, except, pytest.skip, or any construct that lets the criterion pass without

testing what it claims to test;

  • deleting a criterion, renaming its id, or altering its Intent: line;
  • adding a Suite: or Requires: line that was not already there;
  • adding, correcting, or removing Probe:/Probe Count: for any fault other than

missing-selection-probe, invalid-selection-probe, empty-selection, or probe-failed;

  • touching a criterion that was not named in its own block;
  • editing any other part of a specification.

If a criterion asserts something you believe is wrong about the product, repair it anyway and say nothing. A criterion that runs and fails is useful evidence. A criterion that cannot run is none. Build-time repair handles the rest.

Impossible repair

If a criterion cannot be made runnable without changing what it asserts, do not change it. Emit its block unmodified and add one line immediately after the block:

REPAIR_IMPOSSIBLE: <file> :: <check-id> — <one sentence saying what the criterion would have to change>

The separator between the id and the reason is always an em dash (—); filenames and check_ids commonly contain hyphens, so no other character can be used to tell them apart.

Output

For a Bad acceptance container, emit exactly:

=== BEGIN REPAIRED SPECIFICATION ===
<the complete original specification with only AC delimiter lines corrected>
=== END REPAIRED SPECIFICATION ===

For every defective criterion in the batch, emit one block, and nothing else:

Use no prose, no summary, and no fenced code around a criterion block. Use the exact delimiters, with the criterion's own file and id, copied exactly from its file= and check_id= attributes — this is what tells two criteria that happen to share a check_id in different files apart:

=== AC <file> :: <check-id> ===
Intent: <unchanged>

<the repaired Python>
=== END AC <file> :: <check-id> ===

Identify the criterion by its file and id; you do not need to know or reproduce its file's own internal delimiter format. Include the whole criterion — every line that was inside its block — not a patch or a diff.

Acceptance persistence lifecycle

Acceptance store lifecycle. The DATABASE step provisions and verifies the configured store once. Criteria never replace DATABASE_URL/DATABASE_PATH, create a temporary database, apply DDL, or assume an empty store. The store may hold data before any criterion runs: assert on records the criterion created, or on a count read before setup (after == before + 5), never on an absolute table count.

The acceptance harness owns isolation. It begins one transaction before each Python criterion and rolls it back when the criterion ends, pass or fail. Every database the criterion's process opens joins it: db_session(), the application behind web_client(), and commands run through cli_runner(). Setup, behavior and readback therefore see one another's writes, and none persists; leaving a with db_session() as db: block rolls nothing back. Seed with with db_session() as db: and pass that database into feature operations. Never commit, clean up, or open a transaction.

A criterion that launches a process (subprocess, a server on a port, a shell script) cannot see that transaction. It does not open the store and does not depend on rows it wrote; exercise stored behavior in-process instead. Go criteria use OpenSession(ctx, "") with checked deferred cleanup, which rolls back its own transaction. Go HTTP, CLI and restart criteria use OpenCommittedSession(ctx, "") with a criterion-owned identifier and checked defer cleanup through supported product interfaces. Database acceptance must not be replaced by a project test count: runtime catalog and row-count evidence proves provisioning, feature criteria prove behavior.
